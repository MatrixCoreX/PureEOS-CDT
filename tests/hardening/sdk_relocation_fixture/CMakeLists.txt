cmake_minimum_required(VERSION 3.20)
project(pureeos_sdk_relocation LANGUAGES CXX)
find_package(cdt CONFIG REQUIRED)
get_filename_component(actual_sdk "${CDT_ROOT}" REALPATH)
get_filename_component(expected_sdk "${PUREEOS_EXPECTED_SDK}" REALPATH)
if(NOT actual_sdk STREQUAL expected_sdk)
  message(FATAL_ERROR "CDT resolved outside the relocated SDK")
endif()
foreach(tool CMAKE_CXX_COMPILER WASM_LINKER)
  get_filename_component(actual_tool "${${tool}}" REALPATH)
  string(FIND "${actual_tool}" "${expected_sdk}/bin/" prefix_position)
  if(NOT prefix_position EQUAL 0)
    message(FATAL_ERROR "${tool} resolved outside the relocated SDK")
  endif()
endforeach()
if(NOT EXISTS "${PUREEOS_TEST_CONTRACT}")
  message(FATAL_ERROR "Explicit actual contract fixture required")
endif()
add_contract(cdtcompat cdtcompat "${PUREEOS_TEST_CONTRACT}")
